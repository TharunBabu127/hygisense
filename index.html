Arduino C++ Sketch (Copy & Flash): 
#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <HTTPClient.h>

// --- WiFi Credentials ---
const char* WIFI_SSID     = "YOUR_WIFI_NAME";
const char* WIFI_PASSWORD = "YOUR_WIFI_PASSWORD";

// --- Firebase Realtime Database URL ---
const char* FIREBASE_HOST = "https://hygiesense-default-rtdb.asia-southeast1.firebasedatabase.app";
const char* RESTROOM_ID   = "Restroom_1";
const int   DIRTY_THRESHOLD = 3;

// --- Hardware Pins (Internal Pullup Active LOW) ---
#define PIN_CLEAN_BTN 12   // Green Button (Clean / Service Complete)
#define PIN_DIRTY_BTN 14   // Red Button (Dirty feedback)
#define PIN_LED       2    // Built-in Status LED

int cachedGreen = 0;
int cachedDirty = 0;

void fetchLatestCounts() {
  if (WiFi.status() != WL_CONNECTED) return;
  WiFiClientSecure client;
  client.setInsecure();
  HTTPClient https;
  String url = String(FIREBASE_HOST) + "/" + String(RESTROOM_ID) + ".json";
  if (https.begin(client, url)) {
    if (https.GET() == HTTP_CODE_OK) {
      String body = https.getString();
      int gIdx = body.indexOf("\"GreenCount\":");
      if (gIdx != -1) cachedGreen = body.substring(gIdx + 13).toInt();
      int rIdx = body.indexOf("\"RedCount\":");
      if (rIdx != -1) cachedDirty = body.substring(rIdx + 11).toInt();
    }
    https.end();
  }
}

void sendPatch(String json) {
  if (WiFi.status() != WL_CONNECTED) return;
  WiFiClientSecure client;
  client.setInsecure();
  HTTPClient https;
  String url = String(FIREBASE_HOST) + "/" + String(RESTROOM_ID) + ".json";
  if (https.begin(client, url)) {
    https.addHeader("Content-Type", "application/json");
    https.PATCH(json);
    https.end();
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_CLEAN_BTN, INPUT_PULLUP);
  pinMode(PIN_DIRTY_BTN, INPUT_PULLUP);
  pinMode(PIN_LED, OUTPUT);

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connecting WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected! IP: " + WiFi.localIP().toString());
  fetchLatestCounts();
}

void loop() {
  // Green Button (CLEAN) -> Resets Dirty Count to 0 & Sets Optimal
  if (digitalRead(PIN_CLEAN_BTN) == LOW) {
    delay(50); // Debounce
    if (digitalRead(PIN_CLEAN_BTN) == LOW) {
      fetchLatestCounts();
      cachedGreen++;
      cachedDirty = 0; // Strictly reset to 0!
      Serial.println("Clean Pressed! Dirty counter reset to 0.");
      String json = "{\"GreenCount\":" + String(cachedGreen) + ",\"RedCount\":0,\"status\":\"optimal\",\"lastCleaned\":{\".sv\":\"timestamp\"},\"lastUpdated\":{\".sv\":\"timestamp\"}}";
      sendPatch(json);
      while (digitalRead(PIN_CLEAN_BTN) == LOW) delay(10);
    }
  }

  // Red Button (DIRTY / NOT CLEAN) -> Increments towards threshold (3)
  if (digitalRead(PIN_DIRTY_BTN) == LOW) {
    delay(50); // Debounce
    if (digitalRead(PIN_DIRTY_BTN) == LOW) {
      fetchLatestCounts();
      cachedDirty++; // Increments from 0 to 1 after Clean!
      Serial.printf("Dirty Pressed! Count: %d\n", cachedDirty);
      String status = (cachedDirty >= DIRTY_THRESHOLD) ? "needs_attention" : "optimal";
      String json = "{\"RedCount\":" + String(cachedDirty) + ",\"status\":\"" + status + "\",\"lastUpdated\":{\".sv\":\"timestamp\"}}";
      sendPatch(json);
      while (digitalRead(PIN_DIRTY_BTN) == LOW) delay(10);
    }
  }
  delay(20);
}
